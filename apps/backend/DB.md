USE PCStore;

-- ===== CONFIGURACIÓN GLOBAL =====
CREATE TABLE Configuracion (
Id INT IDENTITY(1,1) PRIMARY KEY,
Clave NVARCHAR(50) NOT NULL UNIQUE,
Valor DECIMAL(18,4) NOT NULL,
FechaActualizacion DATETIME NOT NULL DEFAULT GETDATE()
);

-- ===== CLIENTES =====

CREATE TABLE Clientes (
Id INT IDENTITY(1,1) PRIMARY KEY,
Nombre NVARCHAR(150) NOT NULL,
Cuit NVARCHAR(20) NOT NULL UNIQUE,
CondicionIva NVARCHAR(50) NOT NULL,
CondicionFacturacion INT NOT NULL,
Localidad NVARCHAR(100) NOT NULL,
Direccion NVARCHAR(200) NOT NULL,
Telefono NVARCHAR(50) NOT NULL,
Email NVARCHAR(150) NOT NULL,
Observaciones NVARCHAR(MAX) NULL,
Saldo DECIMAL(18,2) NOT NULL DEFAULT 0.00,
Activo BIT NOT NULL DEFAULT 1,
FechaCreacion DATETIME NOT NULL DEFAULT GETDATE()
);
GO

-- ===== CATÁLOGOS =====
CREATE TABLE Marcas (
Id INT IDENTITY(1,1) PRIMARY KEY,
Nombre NVARCHAR(100) NOT NULL,
Descripcion NVARCHAR(255) NULL,
SitioWeb NVARCHAR(255) NULL,
Activo BIT NOT NULL DEFAULT 1
);
GO

CREATE TABLE Proveedores (
Id INT IDENTITY(1,1) PRIMARY KEY,
Nombre NVARCHAR(150) NOT NULL,
Contacto NVARCHAR(100),
Telefono NVARCHAR(50),
Email NVARCHAR(100),
Activo BIT NOT NULL DEFAULT 1
);
GO

CREATE TABLE UnidadesMedida (
Id INT IDENTITY(1,1) PRIMARY KEY,
Nombre NVARCHAR(50) NOT NULL,
Activo BIT NOT NULL DEFAULT 1
);
GO

CREATE TABLE Categorias (
Id INT IDENTITY(1,1) PRIMARY KEY,
Nombre NVARCHAR(100) NOT NULL,
Descripcion NVARCHAR(255) NULL,
Activo BIT NOT NULL DEFAULT 1
)

-- ===== USUARIOS =====
CREATE TABLE Usuarios (
Id INT IDENTITY(1,1) PRIMARY KEY,
NombreUsuario NVARCHAR(50) NOT NULL UNIQUE,
PasswordHash NVARCHAR(255) NOT NULL,
Rol NVARCHAR(30) NOT NULL DEFAULT 'vendedor',
Activo BIT NOT NULL DEFAULT 1,
FechaCreacion DATETIME NOT NULL DEFAULT GETDATE()
);
GO

-- ===== PRODUCTOS =====
CREATE TABLE Productos (
Id INT IDENTITY(1,1) PRIMARY KEY,
CodigoInterno NVARCHAR(50) NOT NULL UNIQUE,
CodigoBarra NVARCHAR(50) NULL,
Descripcion NVARCHAR(255) NOT NULL,
MarcaId INT NOT NULL FOREIGN KEY REFERENCES Marcas(Id),
ProveedorId INT NOT NULL FOREIGN KEY REFERENCES Proveedores(Id),
UnidadId INT NOT NULL FOREIGN KEY REFERENCES UnidadesMedida(Id),
CategoriaId INT NOT NULL FOREIGN KEY REFERENCES Categorias(Id),
Costo DECIMAL(18,2) NOT NULL DEFAULT 0,
CostoDolar DECIMAL(18,2) NOT NULL DEFAULT 0,
IVA DECIMAL(5,2) NOT NULL DEFAULT 21,
Ganancia DECIMAL(5,2) NOT NULL DEFAULT 0,
Precio DECIMAL(18,2) NOT NULL DEFAULT 0,
Stock INT NOT NULL DEFAULT 0,
-- Indica si este producto se maneja con números de serie individuales
ManejaSeries BIT NOT NULL DEFAULT 0,
Activo BIT NOT NULL DEFAULT 1,
FechaCreacion DATETIME NOT NULL DEFAULT GETDATE()
);
GO

CREATE TABLE ProductoCaracteristicas (
Id INT IDENTITY(1,1) PRIMARY KEY,
ProductoId INT NOT NULL FOREIGN KEY REFERENCES Productos(Id) ON DELETE CASCADE,
Clave NVARCHAR(50) NOT NULL,
Valor NVARCHAR(150) NOT NULL
);
GO

CREATE TABLE ProductoImagenes (
Id INT IDENTITY(1,1) PRIMARY KEY,
ProductoId INT NOT NULL FOREIGN KEY REFERENCES Productos(Id) ON DELETE CASCADE,
RutaArchivo NVARCHAR(255) NOT NULL,
EsPrincipal BIT NOT NULL DEFAULT 0
);
GO

-- ===== USUARIOS (ya definido arriba, mov depende de él) =====

-- ===== MOVIMIENTOS DE STOCK =====
CREATE TABLE Movimientos (
Id INT IDENTITY(1,1) PRIMARY KEY,
ProductoId INT NOT NULL FOREIGN KEY REFERENCES Productos(Id),
Tipo NVARCHAR(20) NOT NULL, -- Entrada, Salida, Ajuste
Cantidad INT NOT NULL,
Fecha DATETIME NOT NULL DEFAULT GETDATE(),
Referencia NVARCHAR(100) NULL, -- ej: 'Venta #12', 'Pedido #5'
UsuarioId INT NOT NULL FOREIGN KEY REFERENCES Usuarios(Id),
-- Datos propios de una salida por venta (NULL en entradas/ajustes)
NumeroFactura NVARCHAR(50) NULL,
TipoVenta NVARCHAR(30) NULL, -- ej: Contado, CuentaCorriente, Transferencia
Cliente NVARCHAR(150) NULL
);
GO

-- ===== SERIES (números de serie ligados a un movimiento puntual) =====
CREATE TABLE Series (
Id INT IDENTITY(1,1) PRIMARY KEY,
MovimientoId INT NOT NULL FOREIGN KEY REFERENCES Movimientos(Id) ON DELETE CASCADE,
ProductoId INT NOT NULL FOREIGN KEY REFERENCES Productos(Id),
NumeroSerie NVARCHAR(100) NOT NULL,
Activo BIT NOT NULL DEFAULT 1,
NumeroFactura NVARCHAR(50) NULL,
FechaCreacion DATETIME NOT NULL DEFAULT GETDATE(),
CONSTRAINT UQ_Series_Producto_Numero UNIQUE (ProductoId, NumeroSerie)
);
GO

-- ===== VENTAS =====
CREATE TABLE Ventas (
Id INT IDENTITY(1,1) PRIMARY KEY,
Fecha DATETIME NOT NULL DEFAULT GETDATE(),
UsuarioId INT NOT NULL FOREIGN KEY REFERENCES Usuarios(Id),
Total DECIMAL(18,2) NOT NULL DEFAULT 0,
Activo BIT NOT NULL DEFAULT 1
);
GO

CREATE TABLE VentaDetalle (
Id INT IDENTITY(1,1) PRIMARY KEY,
VentaId INT NOT NULL FOREIGN KEY REFERENCES Ventas(Id) ON DELETE CASCADE,
ProductoId INT NOT NULL FOREIGN KEY REFERENCES Productos(Id),
Cantidad INT NOT NULL,
PrecioUnitario DECIMAL(18,2) NOT NULL
);
GO

-- ===== PEDIDOS A PROVEEDORES =====
CREATE TABLE Pedidos (
Id INT IDENTITY(1,1) PRIMARY KEY,
Fecha NVarchar(50) NOT NULL DEFAULT GETDATE(),
ProveedorId INT NOT NULL FOREIGN KEY REFERENCES Proveedores(Id),
Estado NVARCHAR(30) NOT NULL DEFAULT 'Pendiente',
Activo BIT NOT NULL DEFAULT 1
);
GO

CREATE TABLE PedidoDetalle (
Id INT IDENTITY(1,1) PRIMARY KEY,
PedidoId INT NOT NULL FOREIGN KEY REFERENCES Pedidos(Id) ON DELETE CASCADE,
ProductoId INT NOT NULL FOREIGN KEY REFERENCES Productos(Id),
Cantidad INT NOT NULL,
CostoUnitario DECIMAL(18,2) NOT NULL
);
GO

-- ===== MEDIOS Y TIPOS DE PAGO =====
CREATE TABLE TipoTarjetas (
Id INT IDENTITY(1,1) PRIMARY KEY,
Nombre NVARCHAR(50) NOT NULL UNIQUE, -- ej: Visa, Mastercard, Visa Débito, etc.
Activo BIT NOT NULL DEFAULT 1,
FechaCreacion DATETIME NOT NULL DEFAULT GETDATE()
);
GO

CREATE TABLE MetodosPago (
Id INT IDENTITY(1,1) PRIMARY KEY,
Fecha DATETIME NOT NULL DEFAULT GETDATE(),
Monto DECIMAL(18,2) NOT NULL,
Tipo NVARCHAR(50) NOT NULL,
NroComp NVARCHAR(50) NULL,
ComprobanteId INT NULL,
TipoComprobante NVARCHAR(50) NULL,
Activo BIT NOT NULL DEFAULT 1
);
GO
